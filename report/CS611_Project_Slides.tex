\begin{frame}{Problem, Objectives, and Literature}
\begin{block}{Problem}
MST connects all vertices with minimum cost [1]. Naive Bor\r{u}vka implementations appear 6 to 8$\times$ slower than Kruskal [2,3], but this is caused by missing graph contraction and interpreter overhead, not algorithmic inferiority [4].
\end{block}
\vspace{0.2em}
\textbf{Objectives:} Fix Bor\r{u}vka with contraction + Numba JIT [4,10], evaluate on SNAP benchmarks [9].

% 3/10: ALGORITHMS
\begin{frame}{Algorithm Design: Kruskal vs Bor\r{u}vka [1,2,3]}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Kruskal's (Sequential) [2]}
\begin{enumerate}
    \item Sort all $E$ edges by weight [2]
    \item Scan once: add edge if no cycle [1]
    \item Union-Find: $O(\alpha(V))$ per op [1]
\end{enumerate}
\textbf{Complexity:} $O(E \log E)$ [1]\\
\textbf{Strength:} Sort once, scan once [2]\\
\textbf{Weakness:} Sort is inherently serial [4]
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Bor\r{u}vka's (Parallel) [3]}
\textbf{Each round} ($\leq \log_2 V$ rounds) [1]:
\begin{enumerate}
    \item {\color{kaugreen}Parallel:} Find min edge per component [4]
    \item Add selected edges to MST [3]
    \item {\color{clr2}Contract:} Remove internal edges [4]
\end{enumerate}
\textbf{Key:} Without contraction, scans all $E$ edges $\log V$ times [4]. With contraction, edge set shrinks exponentially [4,6].
\end{block}
\end{column}
\end{columns}
\end{frame}

% 4/10: OPTIMIZATION
\begin{frame}{The Two Optimizations: Contraction + Numba JIT [4,10]}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Graph Contraction [4]}
\begin{itemize}
    \item After each round, remove edges where both endpoints are in the same component [4]
    \item Edge count decreases by $\geq$50\% per round [4]
    \item Eliminates $O(\log V)$ factor of redundant scanning [4,6]
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Numba JIT Compilation [10]}
\begin{itemize}
    \item Compiles Python to machine code via LLVM [10]
    \item Eliminates interpreter overhead ($\sim$10$\times$) [10]
    \item \texttt{prange} enables OpenMP-style multi-threading [10]
    \item Bypasses Python's GIL [10]
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{0.3em}
\begin{block}{Combined Effect on roadNet-CA (100K nodes)}
Python Bor\r{u}vka: \textbf{1.066s}. \quad Numba + Contraction Bor\r{u}vka: \textbf{0.013s}. \quad \textbf{82$\times$ faster.}
\end{block}
\end{frame}

% 9/10: CONCLUSION
\begin{frame}{Conclusion and Recommendations}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Key Findings}
\begin{itemize}
    \item Contraction + JIT: Bor\r{u}vka goes from 6.6$\times$ slower to \textbf{competitive or faster} than Kruskal [2,3,4]
    \item 82$\times$ improvement from two optimizations [4,10]
    \item Full SNAP datasets tractable [9]
    \item Graph structure matters: road (deg 3.1) vs Amazon (deg 6.1) [9]
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Recommendations}
\begin{itemize}
    \item \textbf{Sequential:} Kruskal for prototyping [2]
    \item \textbf{Production:} Bor\r{u}vka with contraction + compiled execution [4,6]
    \item \textbf{Multi-core:} Numba \texttt{prange} or C++ OpenMP [4,10]
    \item \textbf{GPU:} CUDA for 29$\times$ speedup [6,8]
\end{itemize}
\end{block}
\end{column}
\end{columns}
